System.Action tick = () => {
    if (finished) return;
    ticks++;
    var reason = interruption(); if (reason != null) { finish(reason); return; }
    if (phase == "wait_result") {
        sawUsing = sawUsing || activity.IsUsing;
        if (step.Value > baseline && (throwObserved && throwResult || allTargets.Any(p => p == null || p.IsTaken))) { finish("collected"); return; }
        if (throwObserved && !throwResult && !activity.IsUsing) { finish("miss"); return; }
        if (!activity.IsUsing && UnityEditor.EditorApplication.timeSinceStartup - clickAt > .5) { finish(throwObserved && throwResult ? "other_item_collected" : sawUsing ? "miss_without_result_event" : "click_no_action"); return; }
        return;
    }
    if (phase == "release_button") { releaseButton(); phase = "wait_result"; return; }
    if (adVisible()) { finish("ad_overlay"); return; }
    if (unexpectedWindow() || (bool)test.GetType().GetField("_isTest", flags).GetValue(test)) { finish("ui_overlay"); return; }
    if (target == null || target.IsTaken || !target.gameObject.activeInHierarchy) { finish("target_changed"); return; }
    if ((bool)movement.GetType().GetField("_isRotaion", flags).GetValue(movement)) return;
    var collider = target.GetComponentInChildren<UnityEngine.Collider>();
    var aim = collider != null && collider.enabled && collider.gameObject.activeInHierarchy ? collider.bounds.center : target.transform.position;
    var vector = aim - movement.CameraPosition;
    if (vector.magnitude > 12 || vector.magnitude < .1f) { finish("range_outside_skill"); return; }
    var forward = movement.CameraForward.normalized;
    yawError = UnityEngine.Vector3.SignedAngle(UnityEngine.Vector3.ProjectOnPlane(forward, UnityEngine.Vector3.up), UnityEngine.Vector3.ProjectOnPlane(vector, UnityEngine.Vector3.up), UnityEngine.Vector3.up);
    pitchError = Mathf.Atan2(vector.y, new UnityEngine.Vector2(vector.x, vector.z).magnitude) * Mathf.Rad2Deg - Mathf.Asin(Mathf.Clamp(forward.y,-1,1)) * Mathf.Rad2Deg;
    if (ballistic) {
        var fromHand = aim - hand.position;
        var horizontal = new UnityEngine.Vector2(fromHand.x, fromHand.z).magnitude;
        var discriminant = launchSpeedSquared * launchSpeedSquared - gravity * (gravity * horizontal * horizontal + 2 * fromHand.y * launchSpeedSquared);
        if (horizontal < .1f || discriminant < 0) { finish("no_ballistic_solution"); return; }
        var launchElevation = Mathf.Atan((launchSpeedSquared - Mathf.Sqrt(discriminant)) / (gravity * horizontal));
        var desiredElevation = launchElevation - Mathf.Atan2(.25f, dropSpeed);
        pitchError = desiredElevation * Mathf.Rad2Deg - Mathf.Asin(Mathf.Clamp(forward.y,-1,1)) * Mathf.Rad2Deg;
    }
    var yawMotion = Mathf.DeltaAngle(lastYaw, movement.Yaw); var pitchMotion = movement.Pitch - lastPitch;
    if (phase == "settle") {
        settleFrames++;
        stableFrames = Mathf.Abs(yawMotion) < .05f && Mathf.Abs(pitchMotion) < .05f ? stableFrames + 1 : 0;
        lastYaw = movement.Yaw; lastPitch = movement.Pitch;
        if (settleFrames < 3 || stableFrames < 2) return;
        if (Mathf.Abs(yawError) > 1.5f || Mathf.Abs(pitchError) > 1.5f) { phase = "aim"; settleFrames = 0; stableFrames = 0; }
        else {
            if (mode == "aim") { finish("aimed"); return; }
            if (activity.CurrentActivity != hook || activity.IsUsing) { finish("action_changed"); return; }
            if (Survival.Drawers.ArmsDrawer.IsAnimate || button.BlockButton) return;
            var bp = screenCenter(button);
            buttonPointer = new UnityEngine.EventSystems.PointerEventData(es) { position = bp, pressPosition = bp, button = UnityEngine.EventSystems.PointerEventData.InputButton.Left,
                pointerId = -74, eligibleForClick = true, clickCount = 1, clickTime = UnityEngine.Time.unscaledTime };
            var buttonHits = new System.Collections.Generic.List<UnityEngine.EventSystems.RaycastResult>(); es.RaycastAll(buttonPointer, buttonHits);
            if (buttonHits.Count == 0 || !buttonHits[0].gameObject.transform.IsChildOf(button.transform)) { finish("action_covered"); return; }
            buttonPointer.pointerCurrentRaycast = buttonHits[0]; buttonPointer.pointerPressRaycast = buttonHits[0];
            buttonPointer.pointerPress = button.gameObject; buttonPointer.rawPointerPress = buttonHits[0].gameObject;
            down = true; clickFrame = UnityEngine.Time.frameCount; clickAt = UnityEditor.EditorApplication.timeSinceStartup;
            history.Add(new { phase = "click", frame = clickFrame, seconds = Math.Round(clickAt - began,3), yaw = yawError, pitch = pitchError });
            UnityEngine.EventSystems.ExecuteEvents.Execute(button.gameObject, buttonPointer, UnityEngine.EventSystems.ExecuteEvents.pointerDownHandler);
            sawUsing = activity.IsUsing;
            phase = "release_button";
            return;
        }
    }
    if (phase == "aim") {
        aimFrames++;
        if (aimFrames > 120) { finish("aim_not_converging"); return; }
        if (Mathf.Abs(lastDx) > 1 && Mathf.Abs(yawMotion) > .01f) gainX = Mathf.Clamp(Mathf.Abs(yawMotion / lastDx), baseGain * .25f, baseGain * 8);
        if (Mathf.Abs(lastDy) > 1 && Mathf.Abs(pitchMotion) > .01f) gainY = Mathf.Clamp(Mathf.Abs(pitchMotion / lastDy), baseGain * .25f, baseGain * 8);
        lastYaw = movement.Yaw; lastPitch = movement.Pitch;
        if (Mathf.Abs(yawError) < .7f && Mathf.Abs(pitchError) < .7f) {
            releaseLook(); phase = "settle"; lastDx = 0; lastDy = 0; return;
        }
        lastDx = Mathf.Clamp(yawError * .65f / gainX, -120,120);
        lastDy = Mathf.Clamp(pitchError * .65f / gainY, -120,120);
        pointer.delta = new UnityEngine.Vector2(lastDx,lastDy); pointer.position += pointer.delta;
        if (!dragging) {
            UnityEngine.EventSystems.ExecuteEvents.Execute(touch.gameObject, pointer, UnityEngine.EventSystems.ExecuteEvents.initializePotentialDrag);
            UnityEngine.EventSystems.ExecuteEvents.Execute(touch.gameObject, pointer, UnityEngine.EventSystems.ExecuteEvents.beginDragHandler);
            dragging = true;
        }
        UnityEngine.EventSystems.ExecuteEvents.Execute(touch.gameObject, pointer, UnityEngine.EventSystems.ExecuteEvents.dragHandler);
        corrections++;
        if (history.Count < 40) history.Add(new { phase = "look", frame = UnityEngine.Time.frameCount, yaw = yawError, pitch = pitchError, dx = lastDx, dy = lastDy });
    }
    state["result"] = new { status = "running", phase = phase, game_frames = ticks, progress_before = baseline, progress_now = step.Value };
};

}).GetEnumerator());
