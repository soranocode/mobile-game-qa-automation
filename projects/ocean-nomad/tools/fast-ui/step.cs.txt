System.Func<UnityEngine.MonoBehaviour, string> identify = b => {
    var p = path(b.gameObject);
    if (b.name == "Dashboard" && p.Contains("UIHudTopButtonsPanel")) return "open_menu";
    if ((b.name == "UIDashboardButtonSettings" && p.Contains("UIDashboard/")) ||
        (b.name == "Settings" && p.Contains("UIHudTopButtonsPanel"))) return "open_settings";
    if (b.name == "NewGame" && p.Contains("UIPause/")) return "start_new_game";
    if (b.name == "MainMenu" && p.Contains("UIPause/")) return "exit_to_main_menu";
    if (b.name == "UIButtonGreenDark(Clone)" && p.Contains("UIPopupDefault/")) {
        if (newGameConfirmation) return "confirm_new_game";
        if (exitConfirmation) return "confirm_exit";
    }
    if (b.name == "CloseButton" && p.Contains("PremiumNewVariants_16x9_Android/Panel/")) return "close_premium";
    if (b.name == "PremiumButton" && p.EndsWith("PremiumNewVariants_16x9_Android/Panel/Right/Bottom/PremiumButton") &&
        b.GetComponentsInChildren<UnityEngine.UI.Text>(false).Any(t => (t.text ?? "").Trim() == "ПРОДОЛЖИТЬ")) {
        if (premiumPage == "intro_build") return "skip_intro_build";
        if (premiumPage == "intro_survive") return "skip_intro_survive";
    }
    if (b.name == "UIButtonClose" && p.Contains("UIPopupDiscord/Main/UIPopupTitle/Content/")) return "close_discord";
    if (b.name == "SmallButton" && p.EndsWith("MainScreen2/Survival/SmallButton")) return "main_menu_continue";
    return null;
};

            UnityEditor.SessionState.SetString(prefix + "status", "not_sent_target_changed"); return;

        UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject, data, UnityEngine.EventSystems.ExecuteEvents.pointerUpHandler);
