// Close/cancel only via current UI handlers; no window.Hide(), timeScale writes or accept.
__ERROR_WATCH_SETUP__
if(errorWatchBlocked())return new{status="blocked_new_error"};
var windows=UnityEngine.Object.FindObjectsByType<Acme.Gui.Abstractions.AbstractWindowPrefab>(UnityEngine.FindObjectsSortMode.None).Where(w=>w.isActiveAndEnabled&&w.gameObject.scene.IsValid()&&w.gameObject.scene.isLoaded&&w.GetType().FullName!="DamageShower.DamageShowView").ToArray();
if(windows.Length==0)return new{status="clear"};
var es=UnityEngine.EventSystems.EventSystem.current;if(es==null)return new{status="no_event_system"};
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic;
var candidates=new System.Collections.Generic.List<UnityEngine.Component>();
foreach(var w in windows){
    for(var type=w.GetType();type!=null;type=type.BaseType)foreach(var field in type.GetFields(flags|System.Reflection.BindingFlags.DeclaredOnly))
        if(new[]{"_close","close","_closeButton","closeButton","_back","_backButton","_no","_noButton","_cancel","_cancelButton"}.Contains(field.Name)){var c=field.GetValue(w) as UnityEngine.Component;if(c!=null&&c.gameObject.activeInHierarchy&&c is UnityEngine.EventSystems.IPointerClickHandler)candidates.Add(c);}
    candidates.AddRange(w.GetComponentsInChildren<UnityEngine.MonoBehaviour>(false).Where(c=>c is UnityEngine.EventSystems.IPointerClickHandler&&new[]{"Close","CloseButton","UIButtonClose","ButtonClose","Back","BackButton","Cancel","CancelButton","No","NoButton"}.Contains(c.name)));
}
foreach(var target in candidates.Distinct()){
    var simple=target as Survival.Widgets.UI.Button.UISimpleButton;if(simple!=null&&simple.BlockButton)continue;
    var rect=target.GetComponent<UnityEngine.RectTransform>();var canvas=target.GetComponentInParent<UnityEngine.Canvas>();if(rect==null||canvas==null)continue;
    foreach(var offset in new[]{new UnityEngine.Vector2(0,.46f),new UnityEngine.Vector2(-.35f,.46f),UnityEngine.Vector2.zero}){
        var local=rect.rect.center+new UnityEngine.Vector2(rect.rect.width*offset.x,rect.rect.height*offset.y);var point=UnityEngine.RectTransformUtility.WorldToScreenPoint(canvas.renderMode==UnityEngine.RenderMode.ScreenSpaceOverlay?null:canvas.worldCamera,rect.TransformPoint(local));
        var data=new UnityEngine.EventSystems.PointerEventData(es){position=point,pressPosition=point,pointerId=-97,button=UnityEngine.EventSystems.PointerEventData.InputButton.Left,eligibleForClick=true,clickCount=1};
        var hits=new System.Collections.Generic.List<UnityEngine.EventSystems.RaycastResult>();es.RaycastAll(data,hits);if(hits.Count==0||UnityEngine.EventSystems.ExecuteEvents.GetEventHandler<UnityEngine.EventSystems.IPointerClickHandler>(hits[0].gameObject)!=target.gameObject)continue;
        data.pointerCurrentRaycast=hits[0];data.pointerPressRaycast=hits[0];data.pointerPress=target.gameObject;data.rawPointerPress=hits[0].gameObject;
        // Same ordinary UGUI click sequence used by the learned settings route.
        UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject,data,UnityEngine.EventSystems.ExecuteEvents.pointerDownHandler);
        UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject,data,UnityEngine.EventSystems.ExecuteEvents.pointerUpHandler);
        if(errorWatchBlocked())return new{status="blocked_new_error"};
        if(target!=null&&target.gameObject.activeInHierarchy)UnityEngine.EventSystems.ExecuteEvents.Execute(target.gameObject,data,UnityEngine.EventSystems.ExecuteEvents.pointerClickHandler);
        return new{status="closed_one",target=target==null?"destroyed":target.name};
    }
}
return new{status="unknown_or_covered_window",windows=windows.Select(w=>w.GetType().FullName).ToArray()};
