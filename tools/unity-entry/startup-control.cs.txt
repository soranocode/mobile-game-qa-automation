System.Func<UnityEngine.GameObject, string> path = go => {
    var parts = new System.Collections.Generic.List<string>();
    for (var t = go.transform; t != null; t = t.parent) parts.Insert(0, t.name);
    return string.Join("/", parts.ToArray());
};

        owners.AddRange(root.GetComponentsInChildren<UnityEngine.MonoBehaviour>(true));
}

};

                gates[key] = (bool)entry.Value;
